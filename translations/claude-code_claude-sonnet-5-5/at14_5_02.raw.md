mkres <- function(count, denom, events) {
  paste0(count, " (", sprintf("%.1f", count / denom * 100), "%) [", events, "]")
}

fish_one <- function(d) {
  if (length(unique(d$EVENT)) < 2 || length(unique(d$TRTAN)) < 2) return(NULL)
  p <- fisher.test(table(d$TRTAN, d$EVENT))$p.value
  data.frame(Table = "Table TRTAN * EVENT", Name1 = "XP2_FISH",
             Label1 = "Two-sided Pr <= P",
             cValue1 = sprintf("%.10f", p), nValue1 = p,
             stringsAsFactors = FALSE)
}
run_f <- function(df, byvar, trts) {
  d <- df[df$TRTAN %in% trts, , drop = FALSE]
  if (is.null(byvar)) {
    r <- fish_one(d)
    if (is.null(r)) r <- data.frame(Table = character(), Name1 = character(),
                                    Label1 = character(), cValue1 = character(),
                                    nValue1 = numeric())
    return(r)
  }
  keys <- unique(d[[byvar]])
  keys <- keys[order(keys, na.last = FALSE)]
  out <- lapply(keys, function(k) {
    sub <- if (is.na(k)) d[is.na(d[[byvar]]), ] else d[!is.na(d[[byvar]]) & d[[byvar]] == k, ]
    r <- fish_one(sub)
    if (is.null(r)) return(NULL)
    r[[byvar]] <- k
    r[, c(byvar, "Table", "Name1", "Label1", "cValue1", "nValue1")]
  })
  out <- do.call(rbind, out)
  if (is.null(out)) {
    out <- data.frame(x = character(), Table = character(), Name1 = character(),
                      Label1 = character(), cValue1 = character(), nValue1 = numeric())
    names(out)[1] <- byvar
  }
  out
}
