# ---- helpers -----------------------------------------------------------------
# SAS sorts missing values first; radix ordering matches SAS's byte-order collation
sas_sort <- function(df, keys) {
  o <- do.call(order, c(lapply(keys, function(k) df[[k]]),
                        list(na.last = FALSE, method = "radix")))
  df[o, , drop = FALSE]
}

# SAS ROUND-half-away-from-zero to 1 decimal (as used by put(x, 8.1))
sas_round1 <- function(x) sign(x) * floor(abs(x) * 10 + 0.5) / 10

# RESULT = strip(put(count,best.))||' ('||strip(put(pct,8.1))||'%) ['||strip(put(events,best.))||']'
make_result <- function(count, denom, events) {
  pct   <- count / denom * 100
  pct_s <- ifelse(is.na(pct), ".", sprintf("%.1f", sas_round1(pct)))
  ev_s  <- ifelse(is.na(events), ".", as.character(events))
  substr(paste0(as.character(count), " (", pct_s, "%) [", ev_s, "]"), 1, 50)
}

# COUNT(distinct usubjid) by keys
cnt_subj <- function(df, keys) {
  df %>%
    group_by(across(all_of(keys))) %>%
    summarise(count = n_distinct(USUBJID, na.rm = TRUE), .groups = "drop") %>%
    as.data.frame()
}

# COUNT(usubjid) by keys (non-missing rows)
cnt_events <- function(df, keys) {
  df %>%
    group_by(across(all_of(keys))) %>%
    summarise(events = sum(!is.na(USUBJID)), .groups = "drop") %>%
    as.data.frame()
}

# PROC SQL "=" joins never match missing keys -> na_matches = "never"
sas_join <- function(x, y, by) {
  left_join(x, y, by = by, na_matches = "never")
}

# Fisher's exact (two-sided, XP2_FISH) by BY group, mimicking PROC FREQ + ODS OUTPUT
fisher_pairs <- function(dat, trt_levels, byvar = NULL) {
  dat <- dat[!is.na(dat$TRTAN) & dat$TRTAN %in% trt_levels, , drop = FALSE]
  if (is.null(byvar)) {
    groups <- list(dat)
    keys   <- list(NULL)
  } else {
    dat  <- sas_sort(dat, byvar)
    keys <- unique(dat[[byvar]])
    groups <- lapply(keys, function(k) {
      m <- if (is.na(k)) is.na(dat[[byvar]]) else (!is.na(dat[[byvar]]) & dat[[byvar]] == k)
      dat[m, , drop = FALSE]
    })
  }
  kept_keys <- list()
  pvals <- numeric(0)
  for (i in seq_along(groups)) {
    d <- groups[[i]]
    if (nrow(d) == 0) next
    tb <- table(d$TRTAN, d$EVENT)
    if (nrow(tb) != 2 || ncol(tb) != 2) next   # PROC FREQ gives no Fisher table for degenerate tables
    p <- fisher.test(unclass(tb), alternative = "two.sided")$p.value
    kept_keys[[length(kept_keys) + 1]] <- keys[[i]]
    pvals <- c(pvals, p)
  }
  n <- length(pvals)
  res <- data.frame(
    Name1  = rep("XP2_FISH", n),
    Label1 = rep("Two-sided Pr <= P", n),
    cValue1 = rep(NA_character_, n),
    nValue1 = as.numeric(pvals),
    stringsAsFactors = FALSE
  )
  if (!is.null(byvar)) {
    keyvec <- if (n > 0) vapply(kept_keys, as.character, character(1)) else character(0)
    kdf <- data.frame(keyvec, stringsAsFactors = FALSE)
    names(kdf) <- byvar
    res <- cbind(kdf, res)
  }
  res
}

# ---- read inputs (ADAM.ADSL -> data/adsl.xpt, ADAM.ADAE -> data/adae.xpt) ----
read_adam <- function(path) {
  x <- read_xpt(path)
  x <- zap_labels(zap_formats(x))
  names(x) <- toupper(names(x))
  as.data.frame(x)
}

ADSL <- read_adam("data/adsl.xpt")
ADAE <- read_adam("data/adae.xpt")

# ---- descriptive statistics --------------------------------------------------
adsl_s <- ADSL[!is.na(ADSL$SAFFL) & ADSL$SAFFL == "Y", , drop = FALSE]

               !is.na(ADAE$AESER) & ADAE$AESER == "Y", , drop = FALSE]

# CREATE TABLE DENOM
DENOM <- adsl_s %>%
  group_by(TRTAN = TRT01AN) %>%
  summarise(DENOM = n_distinct(USUBJID, na.rm = TRUE), .groups = "drop") %>%
  as.data.frame()
DENOM <- sas_sort(DENOM, "TRTAN")
DENOM <- DENOM[, c("DENOM", "TRTAN")]

# PT level
AEPTD <- sas_sort(cnt_subj(ae_sub, c("TRTAN", "AEBODSYS", "AEDECOD")),
                  c("TRTAN", "AEBODSYS", "AEDECOD"))
AEPTA <- sas_sort(cnt_events(ae_sub, c("TRTAN", "AEBODSYS", "AEDECOD")),
                  c("TRTAN", "AEBODSYS", "AEDECOD"))

RESULTPT <- sas_join(AEPTD, DENOM, by = "TRTAN") %>%
  sas_join(AEPTA, by = c("TRTAN", "AEBODSYS", "AEDECOD")) %>%
  mutate(SUMMARYLEVEL = 3,
         RESULT = make_result(count, DENOM, events)) %>%
  select(count, AEBODSYS, AEDECOD, TRTAN, SUMMARYLEVEL, DENOM, events, RESULT) %>%
  as.data.frame()
RESULTPT <- sas_sort(RESULTPT, c("AEBODSYS", "AEDECOD", "TRTAN"))

# SOC level
AESOCD <- sas_sort(cnt_subj(ae_sub, c("TRTAN", "AEBODSYS")), c("TRTAN", "AEBODSYS"))

RESULTSOC <- sas_join(AESOCD, DENOM, by = "TRTAN") %>%
